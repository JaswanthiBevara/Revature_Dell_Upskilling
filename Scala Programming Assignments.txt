case class Candidate(name:String,score:Option[Int])
object ScoreProcessor {

  def passedCandidates(
                      candidate:Candidate
                      ):Option[String]={
    candidate match{
      case Candidate(name,Some(score)) if score>=60 =>
        Some(name)
      case Candidate(name,Some(score)) =>
        None
      case Candidate(name,None) =>
        None
    }
  }

  def pendingCandidates(
                      candidate:Candidate
                      ):Option[String]={
    candidate match{
      case Candidate(name,None) =>
        Some(name)
      case _ =>
          None
    }
  }

  def failedCandidates(
                    candidate:Candidate
                    ):Option[String]={
    candidate match{
      case Candidate(name,Some(score)) if score<60 =>
        Some(name)
      case _ =>
        None
    }
  }

  def avgScore(
              candidate:Candidate
              ):Option[Int]={
    candidate match{
      case Candidate(name,Some(score)) =>
        Some(score)
      case _=>
        None
    }
  }

  def main(args:Array[String]):Unit={
    val candidates = List(
      Candidate("John", Some(85)),
      Candidate("David", Some(45)),
      Candidate("Smith", None),
      Candidate("Peter", Some(75)),
      Candidate("Alex", None)
    )
    val students_passed = candidates.flatMap(passedCandidates)
    println(s"Passed Candidates: $students_passed")

    val students_pending = candidates.flatMap(pendingCandidates)
    println(s"Pending Candidates: $students_pending")

    val students_failed = candidates.flatMap(failedCandidates)
    println(s"Failed Candidates: $students_failed")

    val scores = candidates.flatMap(avgScore)
    val averageScore:Float = (scores.sum.toFloat/scores.length)
    println(s"Average Score: $averageScore")


  }




}

case class Order(orderId:Int,customerName:String,amount:Option[Double],status:String)

object EOrderProc{

  def valiAmount(
                ord:Order
                ):Option[Int]= {
    ord match {
      case Order(orderId, customerName, Some(amount), status) if amount >0=>
        Some(orderId)
      case _ =>
        None
    }
  }

  def oreCom(
            ord:Order
            ):Option[Order]={
    ord match{
      case Order(orderId, customerName, Some(amount), status) if status=="COMPLETED" =>
        Some(Order(orderId, customerName, Some(amount), status))
      case Order(orderId, customerName, None, status) if status=="COMPLETED" =>
        Some(Order(orderId, customerName, None, status))
      case Order(orderId, customerName, Some(amount), status) =>
        None

    }
  }

  def totalAmount(
                 ord:Order
                 ):Option[Double]={
          ord match {
            case Order(orderId, customerName, Some(amount), status) if amount>0 && status=="COMPLETED" =>
              Some(amount)
            case _ =>
              None
          }
  }

  def penOrd(
              ord:Order
            ):Option[Order]={
    ord match{
      case Order(orderId, customerName, Some(amount), status) if status=="PENDING" =>
        Some(Order(orderId, customerName, Some(amount), status))
      case Order(orderId, customerName, None, status) if status=="PENDING" =>
        Some(Order(orderId, customerName, None, status))
      case _ =>
        None

    }
  }

  def valid(
           ord:Order
           ):Option[(Int,Double)]={
    ord match{
      case Order(orderId, customerName, Some(amount), status)  if amount>0 =>
        Some((orderId,amount))
      case _=>
        None
    }
  }

  def main(args:Array[String]):Unit= {
    val orders = List(
      Order(1, "John", Some(2500), "COMPLETED"),
      Order(2, "David", Some(1500), "PENDING"),
      Order(3, "Smith", None, "COMPLETED"),
      Order(4, "Peter", Some(-500), "FAILED"),
      Order(5, "Alex", Some(4000), "COMPLETED")
    )
    val valamount = orders.flatMap(valiAmount)
    println(s"Valid Amount orders: $valamount")

    val comord = orders.map(oreCom)
    println(s"Comleted orders: $comord")

    val total = orders.flatMap(totalAmount).sum
    println(s"Total Amount of valid and Completed Orders: $total")

    val pen = orders.flatMap(penOrd)
    println(s"Pending Orders: $pen")

    val validOrd=orders.flatMap(valid)
    println(s"Valid Order's Id and Amount: $validOrd")

  }


}

case class Subject1(name: String, marks: Option[Int])
case class Student1(id: Int, name: String, Subjects: List[Subject1])

object StdGradeMang {

  def isIncomplete(student: Student1): Boolean = {
    student match {
      case Student1(id, name, subjects) =>
        subjects.exists {
          case Subject1(subjectName, None) => true
          case Subject1(subjectName, Some(marks)) => false
        }
    }
  }

  def totalMarks(student: Student1): Either[String, Int] = {
    student match {
      case Student1(id, name, subjects) if isIncomplete(student) =>
        Left("Assessments are incomplete")

      case Student1(id, name, subjects) =>
        val total = subjects.flatMap {
          case Subject1(subjectName, Some(marks)) => List(marks)
          case Subject1(subjectName, None) => List.empty[Int]
        }.sum

        Right(total)
    }
  }

  def isPassed(student: Student1): Boolean = {
    student match {
      case Student1(id, name, subjects) =>
        !isIncomplete(student) &&
          subjects.forall {
            case Subject1(subjectName, Some(marks)) => marks >= 40
            case Subject1(subjectName, None) => false
          }
    }
  }

  def hasDistinction(student: Student1): Boolean = {
    totalMarks(student) match {
      case Right(total) =>
        total.toDouble / student.Subjects.size >= 75

      case Left(error) =>
        false
    }
  }

  def main(args: Array[String]): Unit = {

    val students = List(
      Student1(101, "Jashu", List(
        Subject1("Scala", Some(90)),
        Subject1("SQL", Some(85)),
        Subject1("Python", Some(95))
      )),
      Student1(102, "David", List(
        Subject1("Scala", Some(60)),
        Subject1("SQL", Some(70)),
        Subject1("Python", Some(65))
      )),
      Student1(103, "Smith", List(
        Subject1("Scala", Some(80)),
        Subject1("SQL", None),
        Subject1("Python", Some(90))
      )),
      Student1(105, "Alex", List(
        Subject1("Scala", None),
        Subject1("SQL", Some(80)),
        Subject1("Python", None)
      )),
      Student1(106, "Robert", List(
        Subject1("Scala", Some(95)),
        Subject1("SQL", Some(90)),
        Subject1("Python", Some(85))
      ))
    )

    println("STUDENT-WISE TOTAL MARKS")
    students.foreach { student =>
      println(student.name + " : " + totalMarks(student))
    }

    println("\nSTUDENTS WHO PASSED")
    students.filter(isPassed).foreach(println)

    println("\nINCOMPLETE ASSESSMENTS")
    students.filter(isIncomplete).foreach(println)

    println("\nSTUDENTS WITH DISTINCTION")
    students.filter(hasDistinction).foreach(println)
  }
}
