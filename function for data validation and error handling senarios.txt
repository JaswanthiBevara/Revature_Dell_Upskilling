scala> def validateCandidate(
     |     candidate: Candidate
     | ): Either[String, Int] = {
     | candidate match { //Candidate(name, Some(score)) - extracts candidate name and score. Some(score) - confirms that a score exists.
     |
     |     case Candidate(name, Some(score)) if score >= 60 =>
     |       Right(score)
     |
     |     case Candidate(name, Some(score)) =>
     |       Left(s"$name failed with score $score")
     |
     |     case Candidate(name, None) =>
     |       Left(s"$name has no assessment score")
     |   }
     | }
